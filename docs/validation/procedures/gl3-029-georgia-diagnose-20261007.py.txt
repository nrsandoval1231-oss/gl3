import httpx

from gridlens.ingestion.adapters import georgia_air_permits as g
root=Path('.artifacts/regeneration/georgia-identity-diagnosis');root.mkdir(exist_ok=False)
receipts=[]
def get(client,url,limit,params=None,headers=None,name='body'):
 started=datetime.now(UTC).isoformat();h,b=g._bounded_get(client,url,limit,params=params,headers=headers);(root/name).write_bytes(b);receipts.append({'url':url,'params':params,'method':'GET','retrieved_at':started,'body_file':name,'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b),'content_type':h.get('content-type')});(root/'requests.json').write_text(json.dumps(receipts,indent=2));return b
with httpx.Client(timeout=15,follow_redirects=False) as c:

 result={'facility':facility,'location':location,'application':application,'site_key':site,'facility_key':fac,'site_matches':len(site)>=3 and f' {site} ' in f' {normal} ','facility_matches':len(fac)>=5 and f' {fac} ' in f' {normal} ','application_matches':app,'page_count':len(texts),'opening_excerpt':texts[0][:1400],'qts_rows':[{'facility':g._record_value(x,'field_1'),'location':g._record_value(x,'field_2'),'application':g._record_value(x,'field_8')} for x in rows if 'QTS ATL3' in str(x)]};(root/'diagnosis.json').write_text(json.dumps(result,indent=2));print(json.dumps(result,indent=2))
