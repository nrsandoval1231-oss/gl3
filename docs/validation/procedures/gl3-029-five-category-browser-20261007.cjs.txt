const { chromium, expect } = require('/workspace/gl3-regeneration/web/node_modules/@playwright/test');
const fs = require('node:fs');
(async()=>{
const manifest=JSON.parse(fs.readFileSync('.artifacts/regeneration/five-manifest.json','utf8'));
const report=JSON.parse(fs.readFileSync('.artifacts/regeneration/five-validation/report.json','utf8'));
const browser=await chromium.launch(); const page=await browser.newPage();page.setDefaultTimeout(15000);
const checks=[];
try {
 await page.goto('http://127.0.0.1:5238/?view=pursuits');
 await page.getByLabel('Password').fill(process.env.GRIDLENS_TEST_PASSWORD);
 await page.getByRole('button',{name:'Continue',exact:true}).click();
 await expect(page.getByRole('heading',{name:'What should we work today?'})).toBeVisible();
 checks.push('real authenticated pursuit workspace');
 for(const selected of manifest.cases.filter(c=>['chicago_conversion','oil_and_gas','low_quality'].includes(c.category))){
  const priorState=report.cases.find(c=>c.need_id===selected.need_id).before.disposition;
  await page.getByRole('group',{name:'Filter pursuit state'}).getByRole('button',{name:new RegExp('^'+priorState)}).click();
  const card=page.locator(`[data-commercial-need-id="${selected.need_id}"]`);
  await card.locator('.commercial-card-heading').click();
  const panel=card.getByRole('region',{name:'Autonomous pursuit research'});
  await expect(panel.getByRole('heading',{name:'Missing facts · autonomous research'})).toBeVisible();
  const questions=report.cases.find(c=>c.need_id===selected.need_id).questions;
  const occurrences = {};
  for(const q of questions){
   const index=occurrences[q.question]??0; occurrences[q.question]=index+1;
   const article=panel.getByRole('article',{name:`Research question: ${q.question}`,exact:true}).filter({hasText:`Evidence fingerprint: ${q.evidence_fingerprint}`}).first();
   await expect(article).toContainText(q.status.replaceAll('_',' '));
   await article.getByText('What GridLens knew before',{exact:true}).click();
   await expect(article).toContainText(`Saved pursuit state: ${priorState}`);
   await article.getByText('Research trail · sources and evidence decisions',{exact:true}).click();
   await expect(article).toContainText('Sources actually checked');
   await expect(article).toContainText('Model calls: 0');
   if(selected.category==='oil_and_gas') await expect(article).toContainText('No live source refresh');
   expect(await article.locator('a[href^="https://"]').count()).toBeGreaterThan(0);
  }
  await page.screenshot({path:`.artifacts/regeneration/${selected.category}-desktop-five.png`,fullPage:true});
  checks.push(`${selected.category}: actual questions/status/before-state/source-links/budgets`);
  await card.locator('.commercial-card-heading').click();
 }
 await page.setViewportSize({width:390,height:844});await page.reload();
 await expect(page.getByRole('heading',{name:'What should we work today?'})).toBeVisible();
 await page.screenshot({path:'.artifacts/regeneration/fresh-mobile-five.png',fullPage:true});
 checks.push('mobile reload preserves authenticated real corpus');
 fs.writeFileSync('.artifacts/regeneration/fresh-browser-result-five.json',JSON.stringify({passed:true,checks,legacy_six_case_acceptance:false,source:'real regenerated database, no API mocks'},null,2));
} catch(e){fs.writeFileSync('.artifacts/regeneration/fresh-browser-result-five.json',JSON.stringify({passed:false,checks,error:String(e),legacy_six_case_acceptance:false},null,2));process.exitCode=1;}finally{await browser.close();}
})();
